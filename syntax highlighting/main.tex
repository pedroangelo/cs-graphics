\noindent Using newly defined environment so we can have several styles.
\begin{itemize}
\item Writing directly.
\begin{pythoncode}
print("Hello, world!")
\end{pythoncode}

\item Importing from file
%% \lstinputlisting[style=pythonstyle]{get_valid_input.py}
\pythoncodefile{get_valid_input.py}

\item Using newtcblisting to add tag
\begin{pythoncodetag}
print("Hello, world!")
\end{pythoncodetag}

\item Importing from file
%% \tcbinputlisting{pythonstyletag,listing file=get_valid_input.py}
\pythoncodetagfile{get_valid_input.py}
\end{itemize}
