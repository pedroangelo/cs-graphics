\usepackage[most]{tcolorbox}
\tcbuselibrary{listings}

\lstdefinestyle{pythonstyle}{
  language=Python,
  backgroundcolor=\color{background},
  basicstyle=\ttfamily\small\color{base},
  breaklines=true,
  frame=single,
  rulecolor=\color{frame},
  xleftmargin=2mm,
  framexleftmargin=1mm,
  xrightmargin=2mm,
  framexrightmargin=1mm,
  showstringspaces=false,
  tabsize=2,
  numbers=none,
  keywordstyle=\color{keyword}\bfseries,
  stringstyle=\color{string},
  commentstyle=\color{comment}\itshape,
  identifierstyle=\color{identifier},
  numberstyle=\color{number},
  numbersep=0.6em,
  columns=fullflexible,
  keepspaces=true,
  % Rounds all 4 corners
  frameround=tttt,
  literate={á}{{\'a}}1 {ã}{{\~a}}1 {ç}{{\c{c}}}1 {é}{{\'e}}1 {ê}{{\^e}}1
}

\lstdefinestyle{pythonstyletag}{
  style=pythonstyle,
  rulecolor=\color{background},
  xleftmargin=0pt,
  framexleftmargin=0pt,
  xrightmargin=0pt,
  framexrightmargin=0pt,
  aboveskip=1pt,
  belowskip=0pt,
}

\tcbset{
  pythonstyletag/.style={
    listing only,
    listing engine=listings,
    listing options={
      style=pythonstyletag
    },
    colback=background,
    colframe=frame,
    %% colbacktitle=background,
    coltitle=comment,
    fonttitle=\scriptsize\ttfamily,
    title={\hfill Python},
    attach title to upper=,
    %% halign title=right,
    %% titlerule=0pt,
    boxrule=0.5pt,
    arc=2pt,
    top=0mm,
    left=1mm,
    right=1mm,
    bottom=0mm,
    before skip=\medskipamount,
    after skip=\medskipamount
  }
}
