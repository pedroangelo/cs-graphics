\usepackage[most]{tcolorbox}
\tcbuselibrary{listings}

\lstdefinestyle{pythonstyle}{
  language=Python,
  backgroundcolor=\color{ghbg},
  basicstyle=\ttfamily\small,
  breaklines=true,
  frame=single,
  rulecolor=\color{ghframe},
  xleftmargin=2mm,
  framexleftmargin=1mm,
  xrightmargin=2mm,
  framexrightmargin=1mm,
  showstringspaces=false,
  tabsize=2,
  numbers=none,
  keywordstyle=\color{ghkeyword}\bfseries,
  stringstyle=\color{ghstring},
  commentstyle=\color{ghcomment}\itshape,
  numbersep=0.6em,
  columns=fullflexible,
  keepspaces=true,
  frameround=tttt, % Rounds all 4 corners
}

\lstdefinestyle{pythonstyletag}{
  style=pythonstyle,
  rulecolor=\color{ghbg},
  xleftmargin=0pt,
  framexleftmargin=0pt,
  xrightmargin=0pt,
  framexrightmargin=0pt,
  aboveskip=1pt,
  belowskip=0pt,
}

\tcbset{
  pythonstyletag/.style={
    listing only,
    listing engine=listings,
    listing options={
      style=pythonstyletag
    },
    colback=ghbg,
    colframe=ghframe,
    %% colbacktitle=ghbg,
    coltitle=ghcomment,
    fonttitle=\scriptsize\ttfamily,
    title={\hfill Python},
    attach title to upper=,
    %% halign title=right,
    %% titlerule=0pt,
    boxrule=0.5pt,
    arc=2pt,
    top=0mm,
    left=1mm,
    right=1mm,
    bottom=0mm,
    before skip=\medskipamount,
    after skip=\medskipamount
  }
}

\newtcblisting{pythoncodetag}{pythonstyletag}

\lstnewenvironment{pythoncode}
{\lstset{style=pythonstyle}}
{}

\newcommand{\pythoncodetagfile}[1]{%
  \tcbinputlisting{
    pythonstyletag,
    listing file=#1
  }%
}
